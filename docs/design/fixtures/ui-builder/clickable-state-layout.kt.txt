package generated.uibuilder

import androidx.compose.foundation.background
import androidx.compose.foundation.clickable
import androidx.compose.foundation.layout.Box
import androidx.compose.foundation.layout.fillMaxSize
import androidx.compose.foundation.layout.padding
import androidx.compose.runtime.Composable
import androidx.compose.ui.Modifier
import androidx.compose.ui.graphics.Color
import androidx.compose.ui.unit.dp

@Composable
fun ClickableStateLayout() {
    val page = androidx.compose.runtime.remember { androidx.compose.runtime.mutableStateOf<kotlin.Int>(10) }
    Box(modifier = Modifier.fillMaxSize().padding(start = 24.dp, top = 24.dp, end = 24.dp, bottom = 24.dp), content = {
        when (page.value) {
            10 -> {
                Box(modifier = Modifier.fillMaxSize().background(color = Color(4284960932L)).clickable(onClick = { page.value = 20 }), content = {})
            }
            20 -> {
                Box(modifier = Modifier.fillMaxSize().background(color = Color(4278224247L)).clickable(onClick = { page.value = 30 }), content = {})
            }
            else -> {
                Box(modifier = Modifier.fillMaxSize().background(color = Color(4281944491L)).clickable(onClick = { page.value = 10 }), content = {})
            }
        }
    })
}
